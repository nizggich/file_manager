int delete_dir(char *path, int path_size) {
  if (!path)
    return -1;

  DIR *root = NULL;

  if (!(root = opendir(path))) {
    return unlink(path);
  }

  struct dirent *fs_ent = NULL;
  struct stat st;
  int res = 1;

  while ((fs_ent = readdir(root)) != NULL) {

    char *name = fs_ent->d_name;
    int type = fs_ent->d_type;

    if (strcmp(name, ".") == 0 || strcmp(name, "..") == 0) {
      continue;
    }

    if (lstat(path, &st) != 0) {
      perror("lstat failed");
      return -1;
    }

    if (S_ISDIR(st.st_mode)) {
      append_path_segment(path, name, path, path_size);
      res = delete_dir(path, path_size);
    } else {
      res = unlink(path);
    }
  }

  if (res) {
    rmdir(path);
  }

  return res;
}
